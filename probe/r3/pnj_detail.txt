<script>
    $(function () {
        var mobileHeaderTop = $('header.m-header').outerHeight(true);

        $('#app-top-search').css('top', mobileHeaderTop);
        $('.search-background-container').css('top', mobileHeaderTop);

        $('#search-box-click').on('click', function () {
            toggleSearchField();
        });

        $('.search-background-container').on('click', function () {
            toggleSearchField();
        })
    });

    function toggleSearchField() {
        if ($('#search-box-click').hasClass('open')) {
            $('#search-box-open').show();
            $('#search-box-close').hide();
            $('#search-box-click').removeClass('open');
            $('.search-section').hide();
            $('.search-background-container').hide();
            scrollAble();
        } else {
            $('#search-box-open').hide();
            $('#search-box-close').show();
            $('#search-box-click').addClass('open');
            $('.search-section').show();
            $('.search-background-container').show();
            scrollDisable();
        }
    }

    function scrollDisable() {
        $('html, body').css({'overflow': 'hidden', 'height': '100vh'});
        $('html, body').addClass('scrollDisable').on('scroll touchmove mousewheel', function (e) {
            e.preventDefault();
            e.stopPropagation();
            return false;
        });
    }

    function scrollAble() {
        $('html, body').css({'overflow': '', 'height': ''});
        $('html, body').removeClass('scrollDisable').off('scroll touchmove mousewheel');
    }
</script>

<script>
var _snackbarTimer = null;

/**
 * @param  {string} message - 표시할 메시지
 * @param  {Object}  [options]
 * @param  {string}  [options.action]   - 액션 링크 텍스트
 * @param  {string}  [options.url]      - 액션 링크 URL
 * @param  {number}  [options.duration] - 표시 시간 ms (기본 3000)
 */
function showSnackbar(message, options) {
    options = options || {};
    var duration = options.duration || 3000;
    var snackbar = document.getElementById('global-snackbar');
    var messageEl = snackbar.querySelector('.global-snackbar-message');
    var actionEl = snackbar.querySelector('.global-snackbar-action');

    if (_snackbarTimer) {
        clearTimeout(_snackbarTimer);
        _snackbarTimer = null;
    }

    snackbar.classList.remove('show');
    void snackbar.offsetWidth;

    messageEl.textContent = message;

    if (options.action && options.url) {
        actionEl.textContent = options.action;
        actionEl.href = options.url;
        actionEl.style.display = '';
    } else {
        actionEl.style.display = 'none';
    }

    var fadeoutDelay = (duration - 300) / 1000;
    snackbar.style.setProperty('--snackbar-fadeout-delay', fadeoutDelay + 's');
    snackbar.classList.add('show');

    _snackbarTimer = setTimeout(function () {
        snackbar.classList.remove('show');
        _snackbarTimer = null;
    }, duration);
}
</script>
